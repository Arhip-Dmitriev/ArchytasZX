ARCHYTASZX DOCUMENTATION
========================

Python Interfaces

Applies to: ArchytasZX 2.0.x
Implementation: every engine module

The Python interfaces are not yet stable between releases.


1. IMPORT CONVENTIONS
---------------------

Engine objects are imported by full module path. The package and sub-package __init__ files
export nothing; from archytaszx import Diagram fails.

    # Layer A: algebra
    from archytaszx.algebra.dimension import Dim, solve, unify_all
    from archytaszx.algebra.phase import Phase, PhaseVector
    from archytaszx.algebra.scalar import Scalar

    # Layer B: diagrams
    from archytaszx.diagram.graph import Diagram, Direction, PortRef
    from archytaszx.diagram.generators import (
        Z_SPIDER, X_SPIDER, FOURIER_BOX, TRIANGLE, TRIANGLE_INVERSE, W_NODE,
        DIM_BINDER, DIM_SPLITTER)
    from archytaszx.diagram.bangbox import (
        Mult, abstract_port_count, abstract_subgraph_count, instantiate_symbol, peel_one)
    from archytaszx.diagram.validate import validate, validate_or_raise
    from archytaszx.diagram.scalable import to_scalable, from_scalable, strip
    from archytaszx.diagram.compare import isomorphic, canonical_key

    # Layer C: rewriting
    from archytaszx.rewrite.rules_library import RULES, lookup_rule, SPIDER_FUSION
    from archytaszx.rewrite.engine import apply, apply_until_fixpoint, toward_normal_form
    from archytaszx.rewrite.normal_form import normal_form
    from archytaszx.rewrite.egraph import EGraph, simplify
    from archytaszx.rewrite import tactics

    # Layer D: semantics and proof
    from archytaszx.semantics.check import compare, score, compare_symbolic
    from archytaszx.semantics.contract_symbolic import contract_symbolic
    from archytaszx.semantics.decide import decide_equal
    from archytaszx.semantics.prove import prove, check_proof
    from archytaszx.semantics.induction import prove_by_induction
    from archytaszx.semantics.certificate import certify, verify, replay
    from archytaszx.semantics import kernel

    # Text input and output
    from archytaszx.repl.parser import parse, parse_dsl, parse_dirac, parse_listing
    from archytaszx.repl.printer import render, render_graph

Each module raises only exceptions of its own hierarchy (ERRORS.txt).


2. WORKED EXAMPLE
-----------------

Build two Z spiders over a symbolic dimension, fuse them, check, certify, and prove the
fusion for every number of copies.

    from archytaszx.algebra.dimension import Dim
    from archytaszx.diagram.generators import Z_SPIDER
    from archytaszx.diagram.graph import Diagram, Direction, PortRef
    from archytaszx.rewrite.match import find_matches
    from archytaszx.rewrite.engine import apply
    from archytaszx.rewrite.rules_library import SPIDER_FUSION

    d = Dim.symbol("d")
    g = Diagram()
    a = g.add_node(Z_SPIDER, input_dims=[], output_dims=[d, d])
    b = g.add_node(Z_SPIDER, input_dims=[d], output_dims=[d, d])
    g.add_wire(PortRef(a, Direction.OUTPUT, 0), PortRef(b, Direction.INPUT, 0))
    g.set_boundary_outputs([PortRef(a, Direction.OUTPUT, 1),
                            PortRef(b, Direction.OUTPUT, 0),
                            PortRef(b, Direction.OUTPUT, 1)])

    result = apply(g, SPIDER_FUSION, find_matches(g)[0])
    len(result.diagram.nodes)                       1
    result.diagram.scalar                           Scalar(1)

    from archytaszx.semantics.check import compare
    from archytaszx.semantics.certificate import certify, verify
    compare(g, result.diagram, {"d": 5}).matched    True
    certificate = certify(g, [result], label="fuse A into B")
    verify(certificate, {"d": 3}).verified          True

    from archytaszx.diagram.bangbox import abstract_subgraph_count
    from archytaszx.semantics.induction import prove_by_induction
    boxed, box_id, mult = abstract_subgraph_count(g, frozenset({a, b}), 1, stem="m")
    fused = apply(boxed, SPIDER_FUSION, find_matches(boxed)[0]).diagram
    proof = prove_by_induction(boxed, fused, witness={"d": 2})
    proof.verdict                                   Verdict.PROVED_UNIFORM
    proof.held_symbolic                             frozenset({'d'})

The layer documents (ALGEBRA.txt, DIAGRAM.txt, REWRITE.txt, SEMANTICS.txt, PROOFS.txt) specify
every public class and function used here.
